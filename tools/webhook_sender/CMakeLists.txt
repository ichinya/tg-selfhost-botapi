# Loaded inside the pinned upstream project after the production target is defined.
get_target_property(api_sources telegram-bot-api SOURCES)
get_target_property(api_libraries telegram-bot-api LINK_LIBRARIES)
set(api_main ${api_sources})
list(FILTER api_main INCLUDE REGEX "telegram-bot-api/telegram-bot-api\\.cpp$")
list(LENGTH api_main main_count)
if(NOT main_count EQUAL 1)
  message(FATAL_ERROR "Upstream source layout changed; review sender adapter")
endif()
list(FILTER api_sources EXCLUDE REGEX "telegram-bot-api/telegram-bot-api\\.cpp$")
list(TRANSFORM api_sources PREPEND "${CMAKE_SOURCE_DIR}/")
add_library(webhook_production_objects OBJECT ${api_sources})
target_include_directories(webhook_production_objects PRIVATE "${CMAKE_SOURCE_DIR}")
target_link_libraries(webhook_production_objects PRIVATE ${api_libraries})
set_property(TARGET telegram-bot-api PROPERTY SOURCES ${api_main} $<TARGET_OBJECTS:webhook_production_objects>)
add_executable(webhook-sender webhook_sender_test.cpp $<TARGET_OBJECTS:webhook_production_objects>)
target_include_directories(webhook-sender PRIVATE "${CMAKE_SOURCE_DIR}")
target_link_libraries(webhook-sender PRIVATE ${api_libraries})
install(TARGETS webhook-sender RUNTIME DESTINATION libexec)
