# Included by the pinned upstream root after its telegram-bot-api target exists.
if(NOT TARGET telegram-bot-api)
  message(FATAL_ERROR "Configure inside the pinned Bot API source tree")
endif()
get_target_property(server_sources telegram-bot-api SOURCES)
get_target_property(server_libraries telegram-bot-api LINK_LIBRARIES)
set(server_main ${server_sources})
list(FILTER server_main INCLUDE REGEX "/telegram-bot-api\\.cpp$")
list(FILTER server_sources EXCLUDE REGEX "/telegram-bot-api\\.cpp$")
list(LENGTH server_main main_count)
if(NOT main_count EQUAL 1)
  message(FATAL_ERROR "Upstream entrypoint changed; review qualification adapter")
endif()
# Both executables use precisely these compiled production objects and TDLib libs.
list(TRANSFORM server_sources PREPEND "${CMAKE_SOURCE_DIR}/")
add_library(release_webhook_core OBJECT ${server_sources})
target_include_directories(release_webhook_core PRIVATE "${CMAKE_SOURCE_DIR}")
target_link_libraries(release_webhook_core PRIVATE ${server_libraries})
foreach(property COMPILE_OPTIONS COMPILE_DEFINITIONS INCLUDE_DIRECTORIES)
  get_target_property(server_property telegram-bot-api ${property})
  if(server_property)
    set_property(TARGET release_webhook_core PROPERTY ${property} ${server_property})
  endif()
endforeach()
set_property(TARGET telegram-bot-api PROPERTY SOURCES ${server_main} $<TARGET_OBJECTS:release_webhook_core>)
add_executable(release-webhook-sender webhook_sender_test.cpp $<TARGET_OBJECTS:release_webhook_core>)
target_include_directories(release-webhook-sender PRIVATE "${CMAKE_SOURCE_DIR}")
target_link_libraries(release-webhook-sender PRIVATE ${server_libraries})
install(TARGETS release-webhook-sender RUNTIME DESTINATION libexec)
